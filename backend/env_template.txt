# ButtonLog Environment Configuration
# Copy this file to .env and fill in your actual values
# Port Allocation: Web 14015 (dev), 14016 (test), DB 15442

# Database Configuration
export DATABASE_PORT=15442

# Google OAuth Configuration
export GOOGLE_CLIENT_ID="your-google-client-id-here"
export GOOGLE_CLIENT_SECRET="your-google-client-secret-here"
export GOOGLE_REDIRECT_URI="http://localhost:14015/auth/google/callback"

# Facebook OAuth (optional)
export FACEBOOK_CLIENT_ID="your-facebook-client-id-here"
export FACEBOOK_CLIENT_SECRET="your-facebook-client-secret-here"

# Apple OAuth (optional)
export APPLE_CLIENT_ID="your-apple-client-id-here"
export APPLE_CLIENT_SECRET="your-apple-client-secret-here"

# Push Notifications - Firebase Cloud Messaging (Android)
# Get these from Firebase Console > Project Settings > Service Accounts
export FCM_PROJECT_ID="your-firebase-project-id"
export FCM_SERVICE_ACCOUNT_JSON='{"type":"service_account","project_id":"...","private_key":"...","client_email":"..."}'
# Or path to service account file (alternative to JSON string)
# export FCM_SERVICE_ACCOUNT_PATH="/path/to/service-account.json"

# Push Notifications - Apple Push Notification Service (iOS)
# Get these from Apple Developer > Certificates, Identifiers & Profiles > Keys
export APNS_KEY_ID="your-apns-key-id"
export APNS_TEAM_ID="your-apple-team-id"
export APNS_KEY_PATH="/path/to/AuthKey_XXXXXX.p8"
# or inline key content
# export APNS_KEY_CONTENT="-----BEGIN PRIVATE KEY-----\n...\n-----END PRIVATE KEY-----"
export APNS_BUNDLE_ID="com.buttonlog.app"
# Use "api.sandbox.push.apple.com" for development, "api.push.apple.com" for production
export APNS_ENVIRONMENT="sandbox"

# AWS SES Email Configuration (Production only)
# These credentials need SendEmail permission in AWS SES
# The from email address must be verified in AWS SES
export AWS_ACCESS_KEY_ID="your-aws-access-key-id"
export AWS_SECRET_ACCESS_KEY="your-aws-secret-access-key"
export AWS_REGION="us-east-1"
export SES_FROM_EMAIL="noreply@yourdomain.com"
export SES_FROM_NAME="ButtonLog"
